FlowCanvas is a reusable package. Feed it your catalog and it builds the node library from your API input — each node may declare a JSON formula.
import { FlowCanvas, defineNodes } from "@flowstock/flow";

const nodes = defineNodes(await api.get("/automation/catalog"));

FlowCanvas.mount("#automations", {
  nodes,
  onRun: (graph) => api.post("/automation/execute", graph),
});
